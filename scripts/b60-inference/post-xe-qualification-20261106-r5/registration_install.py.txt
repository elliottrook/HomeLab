def install_registration(bookings, row):
 """Only called under the existing clear shared host exclusion after News."""
 pilot=BASE.parent/'full-inference-pilot'
 old={n:protected_bytes(pilot/n,executable=True,maximum=262144) for n in REG_PREDECESSORS}
 assert all(hashlib.sha256(old[n]).hexdigest()==sha for n,sha in REG_PREDECESSORS.items())
 assert not (pilot/'b60_post_xe_r5_card_bridge.py').exists()
 old_manifest=json.loads(old['lab_hours_release_manifest.json'])
 for n,sha in old_manifest['pins'].items():
  assert hashlib.sha256(protected_bytes(pilot/n,executable=True,maximum=262144)).hexdigest()==sha
 new={n:base64.b64decode(raw) for n,raw in REG_FILES.items()}
 assert all(hashlib.sha256(new[n]).hexdigest()==REG_SHA[n] for n in new)
 for n,raw in new.items():
  if n.endswith('.py'):compile(raw,str(pilot/n),'exec')
 nm=json.loads(new['lab_hours_release_manifest.json'])
 assert all(nm['pins'][n]==sha for n,sha in old_manifest['pins'].items() if n not in new)
 assert set(nm['pins'])==set(old_manifest['pins'])|{'b60_post_xe_r5_card_bridge.py'}
 before=protected_bytes(ROOT/'reservations/records'/(B+'.json'))
 assert json.loads(before)==row and row['workload'] is None
 backup=ROOT/'registration-backups'/B;backup.parent.mkdir(mode=0o700,exist_ok=True)
 bs=backup.parent.lstat();assert backup.parent.is_dir() and not backup.parent.is_symlink() and bs.st_uid==os.geteuid() and bs.st_mode&0o777==0o700
 assert not backup.exists();backup.mkdir(mode=0o700)
 for n,raw in old.items():create(backup/n,raw)
 create(backup/'booking-before.json',before)
 old_modes={n:os.stat(pilot/n,follow_symlinks=False).st_mode&0o777 for n in old}
 def replace_exact(n,raw,expected,mode):
  path=pilot/n
  if expected is None:assert not path.exists() and not path.is_symlink()
  else:assert hashlib.sha256(protected_bytes(path,executable=True,maximum=262144)).hexdigest()==expected
  tmp=pilot/('.b60-'+B+'-'+n);create(tmp,raw,mode);os.replace(tmp,path)
  fd=os.open(pilot,os.O_RDONLY);os.fsync(fd);os.close(fd)
 try:
  for n in ('b60_post_xe_r5_card_bridge.py','lab_hours_workloads.py','lab_hours_reserved_runner.py','lab_hours_release_manifest.json'):
   replace_exact(n,new[n],REG_PREDECESSORS.get(n),old_modes.get(n,0o600))
  validation="import json,sys;from pathlib import Path;sys.path.insert(0,"+repr(str(pilot))+");import lab_hours_workloads as w;w.validate_registry();import lab_hours_reserved_runner as r;r.load_manifest();v=w.validate_inactive_target(Path("+repr(str(pilot))+"),workload_id='b60-accepted-sycl-post-xe-qualification-v2');print(json.dumps(v))"
  result=json.loads(call(['/usr/bin/python3','-B','-c',validation],30));assert result['effects']=='not-attempted'
 except BaseException:
  # Roll back only our exact installed bytes. Refuse to overwrite an unexpected
  # concurrent writer. No model, claim or consumer effect has occurred here.
  for n in reversed(('b60_post_xe_r5_card_bridge.py','lab_hours_workloads.py','lab_hours_reserved_runner.py','lab_hours_release_manifest.json')):
   path=pilot/n
   if n in old:
    observed=hashlib.sha256(protected_bytes(path,executable=True,maximum=262144)).hexdigest()
    if observed==REG_SHA[n]:replace_exact(n,old[n],REG_SHA[n],old_modes[n])
    else:assert observed==REG_PREDECESSORS[n]
   elif path.exists():
    assert hashlib.sha256(protected_bytes(path,executable=True)).hexdigest()==REG_SHA[n]
    path.unlink()
  assert all(hashlib.sha256(protected_bytes(pilot/n,executable=True,maximum=262144)).hexdigest()==sha for n,sha in REG_PREDECESSORS.items())
  assert not (pilot/'b60_post_xe_r5_card_bridge.py').exists()
  create(backup/'rollback.json',canonical({'schema':'b60-registration-rollback.v1','booking_id':B,'source_restored':True,'model_calls':0}))
  raise
 # Existing atomic/custodied store API: change only the missing workload name,
 # retaining the exact owner, exclusive kind, times, state and all other fields.
 with bookings.locked():
  live=bookings._read_unlocked(bookings._path(B));assert live==row
  bound=dict(live,workload='b60-accepted-sycl-post-xe-qualification-v2')
  bookings._write_unlocked(bound)
 after=protected_bytes(ROOT/'reservations/records'/(B+'.json'))
 receipt={'schema':'b60-registered-exclusive-route.v1','booking_id':B,
  'workload':'b60-accepted-sycl-post-xe-qualification-v2','runner_kind':'b60-post-xe-card-r5',
  'route':'exact-future-clock-to-registered-runner-to-isolated-qualified-owner',
  'booking_before_sha256':hashlib.sha256(before).hexdigest(),'booking_after_sha256':hashlib.sha256(after).hexdigest(),
  'only_booking_field_changed':'workload','owner_times_kind_state_preserved':True,
  'global_file_sha256':REG_SHA,'previous_global_file_sha256':REG_PREDECESSORS,
  'base30_modified':False,'dispatcher_and_reservation_code_modified':False,
  'sole_claimant':'existing-qualified-ExclusiveSession','model_calls':0,
  'installed_epoch':int(time.time())}
 create(backup/'receipt.json',canonical(receipt))
 return receipt
